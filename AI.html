def custom_knn_matcher(des1, des2):
    matches_list = []
    
    # Ensure descriptors are uint8 for bitwise operations
    des1 = des1.astype(np.uint8)
    des2 = des2.astype(np.uint8)

    for i in range(len(des1)):
        # Calculate Hamming Distance: 
        # 1. XOR finds differing bits
        # 2. .unpackbits + .sum counts the 1s efficiently in NumPy
        diff = np.bitwise_xor(des1[i], des2)
        # Summing bits across the 32 bytes of the ORB descriptor
        dist = np.unpackbits(diff, axis=1).sum(axis=1)
        
        # Find indices of the two smallest distances
        idx2 = np.argpartition(dist, 2)[:2] # Faster than sort for just 2 elements, due to only needing to find the two smallest, and not the rest
        idx2 = idx2[np.argsort(dist[idx2])] # Ensure d1 is actually smaller than d2
        
        d1, d2 = dist[idx2[0]], dist[idx2[1]]
        
        # Manually construct OpenCV DMatch objects
        m = cv2.DMatch(i, int(idx2[0]), float(d1))
        n = cv2.DMatch(i, int(idx2[1]), float(d2))
        
        matches_list.append([m, n])
        
    return matches_list
